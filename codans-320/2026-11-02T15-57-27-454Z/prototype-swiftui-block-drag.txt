+import CodansCore
+import SwiftUI
+import UniformTypeIdentifiers
+import OSLog
+
+let pdLog = Logger(subsystem: "com.gumpw.codans", category: "PDRAGDBG")
+
+// Direct drag of a Project block in the sidebar (ADR 0001). A Project block is
+// the Project row plus its worktree rows, which are sibling List rows, so
+// `ForEach.onMove` cannot move it. Each row of a block is a drop target that
+// snaps to the nearest block boundary, and the sidebar draws the insertion
+// indicator at that boundary.
+
+extension UTType {
+  /// Declared in `mac-Info.plist`. A private type: only a Project row drag
+  /// can drop on the sidebar, never text or files from another app.
+  nonisolated static let codansProjectBlock = UTType(exportedAs: "com.gumpw.codans.project-block")
+}
+
+/// The drag payload of a Project row.
+nonisolated struct ProjectBlockDragPayload: Codable, Transferable {
+  let projectID: ProjectID
+
+  static var transferRepresentation: some TransferRepresentation {
+    CodableRepresentation(contentType: .codansProjectBlock)
+  }
+}
+
+/// Where one sidebar row sits: in which visible Project block, and at which
+/// row of that block.
+struct ProjectBlockRowSlot: Equatable {
+  /// Index of the block among the visible Project blocks.
+  let block: Int
+  /// Number of visible Project blocks.
+  let blockCount: Int
+  /// Row index inside the block; the Project row is `0`.
+  let row: Int
+  /// Rows in the block: the Project row plus the rows shown under it.
+  let rowCount: Int
+
+  /// The block boundary nearest to a point `fraction` (0…1) down this row:
+  /// `block` (above the block) or `block + 1` (below it).
+  func boundary(atFraction fraction: Double) -> Int {
+    let position = Double(row) + min(max(fraction, 0), 1)
+    return position < Double(rowCount) / 2 ? block : block + 1
+  }
+
+  /// The row draws the indicator for `boundary` when it is the top row of
+  /// the block below the boundary, or the bottom row of the last block.
+  func indicatorEdge(for boundary: Int?) -> VerticalEdge? {
+    guard let boundary else { return nil }
+    if row == 0, boundary == block { return .top }
+    if row == rowCount - 1, block == blockCount - 1, boundary == blockCount { return .bottom }
+    return nil
+  }
+}
+
+/// The boundary the pointer is over, and the row that reported it. Rows
+/// report enter and exit in either order, so a row clears only its own value.
+struct ProjectBlockDropTarget: Equatable {
+  let boundary: Int
+  let reporter: ProjectBlockRowSlot
+}
+
+private struct ProjectBlockDropDelegate: DropDelegate {
+  let slot: ProjectBlockRowSlot
+  let rowHeight: Double
+  @Binding var target: ProjectBlockDropTarget?
+  let onDrop: (_ projectID: ProjectID, _ boundary: Int) -> Void
+
+  func validateDrop(info: DropInfo) -> Bool {
+    info.hasItemsConforming(to: [.codansProjectBlock])
+  }
+
+  func dropEntered(info: DropInfo) { pdLog.notice("PDRAGDBG enter \(slot.block) \(slot.row) y=\(info.location.y) h=\(rowHeight)"); report(info) }
+
+  func dropUpdated(info: DropInfo) -> DropProposal? {
+    report(info)
+    return DropProposal(operation: .move)
+  }
+
+  func dropExited(info: DropInfo) {
+    pdLog.notice("PDRAGDBG exit \(slot.block) \(slot.row) owner=\(target?.reporter == slot)")
+    if target?.reporter == slot { target = nil }
+  }
+
+  func performDrop(info: DropInfo) -> Bool {
+    let boundary = boundary(info)
+    pdLog.notice("PDRAGDBG perform \(slot.block) \(slot.row) boundary=\(boundary)")
+    target = nil
+    guard let provider = info.itemProviders(for: [.codansProjectBlock]).first else { return false }
+    _ = provider.loadTransferable(type: ProjectBlockDragPayload.self) { result in
+      guard case .success(let payload) = result else { return }
+      Task { @MainActor in onDrop(payload.projectID, boundary) }
+    }
+    return true
+  }
+
+  private func boundary(_ info: DropInfo) -> Int {
+    slot.boundary(atFraction: rowHeight > 0 ? info.location.y / rowHeight : 0)
+  }
+
+  private func report(_ info: DropInfo) {
+    let next = ProjectBlockDropTarget(boundary: boundary(info), reporter: slot)
+    if target != next { pdLog.notice("PDRAGDBG target \(next.boundary) from \(slot.block)/\(slot.row)"); target = next }
+  }
+}

+/// Insertion indicator in the style of the AppKit source-list drop line: an
+/// accent-colored line that starts with a small ring.
+struct ProjectDropIndicator: View {
+  static let ringDiameter: Double = 6
+  static let lineWidth: Double = 2
+
+  var body: some View {
+    HStack(spacing: 0) {
+      Circle()
+        .strokeBorder(Color.accentColor, lineWidth: Self.lineWidth)
+        .frame(width: Self.ringDiameter, height: Self.ringDiameter)
+      Rectangle()
+        .fill(Color.accentColor)
+        .frame(height: Self.lineWidth)
+    }
+    .frame(height: Self.ringDiameter)
+    .allowsHitTesting(false)
+    .accessibilityHidden(true)
+  }
+}

+private struct ProjectBlockDropTargetModifier: ViewModifier {
+  let slot: ProjectBlockRowSlot
+  /// The row's List insets. The indicator sits on the boundary between two
+  /// rows and starts at the sidebar's leading edge, not at the row content.
+  let rowInsets: EdgeInsets
+  @Binding var target: ProjectBlockDropTarget?
+  let onDrop: (ProjectID, Int) -> Void
+  @State private var rowHeight: Double = 0
+
+  func body(content: Content) -> some View {
+    let edge = slot.indicatorEdge(for: target?.boundary)
+    content
+      .onGeometryChange(for: Double.self) { $0.size.height } action: { rowHeight = $0 }
+      .onDrop(
+        of: [.codansProjectBlock],
+        delegate: ProjectBlockDropDelegate(
+          slot: slot, rowHeight: rowHeight, target: $target, onDrop: onDrop)
+      )
+      .overlay(alignment: edge == .bottom ? .bottom : .top) {
+        if let edge {
+          // Center the line on the gap between rows: the Project row's top
+          // inset plus the previous row's bottom inset, or below the last row.
+          let gapCenter = edge == .top ? (rowInsets.top + Self.rowBottomInset) / 2 : rowInsets.bottom / 2
+          let halfHeight = ProjectDropIndicator.ringDiameter / 2
+          ProjectDropIndicator()
+            .padding(.leading, -rowInsets.leading)
+            .offset(y: edge == .top ? -gapCenter - halfHeight : gapCenter + halfHeight)
+        }
+      }
+  }
+
+  /// Bottom inset every sidebar row uses.
+  private static let rowBottomInset: Double = 2
+}
+
+extension View {
+  /// Makes a sidebar row part of a Project block drop target.
+  func projectBlockDropTarget(
+    _ slot: ProjectBlockRowSlot,
+    rowInsets: EdgeInsets,
+    target: Binding<ProjectBlockDropTarget?>,
+    onDrop: @escaping (ProjectID, Int) -> Void
+  ) -> some View {
+    modifier(
+      ProjectBlockDropTargetModifier(slot: slot, rowInsets: rowInsets, target: target, onDrop: onDrop))
+  }
+}
